       CASE
           WHEN b.price > 600 THEN 'Expensive'
           ELSE 'Affordable'
       END AS price_classification
FROM Books b
LEFT JOIN Borrow br
    ON b.book_id = br.book_id
LEFT JOIN Members m
    ON br.member_id = m.member_id;
